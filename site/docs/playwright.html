Playwright

from playwright.sync_api import sync_playwright
with sync_playwright() as p:
    browser = p.chromium.launch(proxy={
        "server": "http://HOST:8899",
        "username": "USER_sess-job42",  # sticky per job
        "password": "PASS",
    })
    page = browser.new_page()
    page.goto("https://api.ipify.org")
    print(page.content())
    browser.close()

One browser context per session id gives clean identity separation. Rotate the session id per job; keep it stable within a job.
