Basic requests

import requests
proxies = {
  "http": "http://USER:PASS@HOST:8899",
  "https": "http://USER:PASS@HOST:8899",
}
r = requests.get("https://api.ipify.org", proxies=proxies, timeout=20)
print(r.status_code, r.text)

Rotation per request

import itertools, secrets
def session_user(base):
    return f"{base}_sess-{secrets.token_hex(4)}"
# new session id per request = fresh exit IP

proxies["http"] = "http://USER_sess-shop1:PASS@HOST:8899"

import time
for attempt in range(4):
    try:
        r = requests.get(url, proxies=proxies, timeout=20)
        if r.status_code == 407:
            raise SystemExit("login expired or wrong — check dashboard, then renew")
        r.raise_for_status()
        break
    except requests.RequestException:
        time.sleep(2 ** attempt)
