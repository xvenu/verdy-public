Node.js setup

fetch with https-proxy-agent

const { HttpsProxyAgent } = require("https-proxy-agent");
const agent = new HttpsProxyAgent("http://USER:PASS@HOST:8899");
const r = await fetch("https://api.ipify.org", { agent });
console.log(r.status, await r.text());

axios

const axios = require("axios");
const r = await axios.get("https://api.ipify.org", {
  proxy: { host: "HOST", port: 8899, auth: { username: "USER", password: "PASS" } },
  timeout: 20000,
});

Notes

Keep credentials in environment variables, never in code. Handle 407 (renew) and 429 (back off) explicitly — see Troubleshooting.
